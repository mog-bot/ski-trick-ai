function updateFolderImportState(){
  const btn=$("importFolderBtn");
  if(!btn) return;

  const files=[...($("folderFiles")?.files || [])].filter(f=>f.type.startsWith("video/"));
  const label=$("folderTrickLabel")?.value.trim() || "";

  const hasVideo = files.length > 0;
  btn.disabled = busy || !label || !hasVideo;
}

function inferSingleFolderLabel(files){
  if(!files.length) return "";
  const rel = files[0].webkitRelativePath || "";
  const parts = rel.split("/").filter(Boolean);
  return parts.length >= 2 ? parts[0].trim() : "";
}

$("importFolderBtn")?.addEventListener("click",async()=>{
  const files=[...$("folderFiles").files].filter(f=>f.type.startsWith("video/"));
  const label=($("folderTrickLabel")?.value.trim() || inferSingleFolderLabel(files));
  const steps=clamp(parseInt($("stepsInput").value)||48,16,64);

  if(!files.length) return alert("Choose a folder containing video files.");
  if(!label) return alert("Enter the trick name at the top of this section before importing the folder.");
  if(!poseLandmarker){
    $("folderStatus").textContent="The body-tracking model is still loading. Wait a moment, then press Upload folder to training again.";
    return;
  }

  setBusy(true);

  let added=0;
  let failed=0;

  for(let i=0;i<files.length;i++){
    const file=files[i];

    try{
      const r=await extractSequence(file,steps,(n,total,found)=>{
        $("folderStatus").textContent=
          `Training class: ${label}\n`+
          `Video ${i+1}/${files.length}: ${file.name}\n`+
          `Pose sample ${n}/${total} · found ${found}/${n}`;
      });

      dataset.push({
        label,
        name:file.name,
        sequence:r.sequence,
        quality:r.avgQuality,
        detectionRate:r.detectionRate,
        steps
      });

      added++;
      renderDataset();

    }catch(err){
      failed++;
      $("folderStatus").textContent=
        `Skipped ${file.name}\n${err.message}`;
    }
  }

  $("folderFiles").value="";

  $("folderStatus").textContent=
    `Imported ${added} video${added===1?"":"s"} into the "${label}" training class.`+
    (failed ? ` ${failed} video${failed===1?" was":"s were"} skipped because tracking was too weak.` : "");

  if(model){
    model.dispose();
    model=null;
    classNames=[];
    $("predictStatus").textContent="Training data changed. Retrain the model.";
    if($("lineStatus")) $("lineStatus").textContent="Training data changed. Retrain the model.";
    if($("versusStatus")) $("versusStatus").textContent="Training data changed. Retrain the model.";
  }

  setBusy(false);
});

window.skiTrainerImportRemote = async function(entries){
  if(!Array.isArray(entries) || !entries.length) throw new Error("No Supabase training videos were supplied.");
  if(!poseLandmarker) throw new Error("The body-tracking model is still loading.");

  const steps=clamp(parseInt($("stepsInput").value)||48,16,64);
  let added=0,failed=0;
  setBusy(true);

  try{
    for(let i=0;i<entries.length;i++){
      const entry=entries[i];
      const label=String(entry.label||"").trim();
      if(!label || !entry.signedUrl){ failed++; continue; }

      try{
        $("trainStatus").textContent=`Supabase ${i+1}/${entries.length} · downloading ${label}/${entry.name}`;
        const response=await fetch(entry.signedUrl,{cache:"no-store"});
        if(!response.ok) throw new Error(`Download failed (${response.status})`);
        const blob=await response.blob();
        const file=new File([blob],entry.name||`training-${i+1}.mp4`,{
          type:entry.mimeType || blob.type || "video/mp4"
        });

        const r=await extractSequence(file,steps,(n,total,found)=>{
          $("trainStatus").textContent=`Supabase ${i+1}/${entries.length} · ${label}/${file.name}\nPose sample ${n}/${total} · found ${found}/${n}`;
        });

        dataset.push({
          label,
          name:`Supabase/${label}/${file.name}`,
          sequence:r.sequence,
          quality:r.avgQuality,
          detectionRate:r.detectionRate,
          steps,
          remote:true
        });
        added++;
        renderDataset();
      }catch(err){
        failed++;
        $("trainStatus").textContent=`Skipped ${label}/${entry.name}: ${err.message}`;
      }
    }

    if(added>0) markDatasetChanged("Supabase training data imported. Train the trick model before testing.");
    $("trainStatus").textContent=`Supabase import finished. Added ${added} clip${added===1?"":"s"}.`+
      (failed?` Skipped ${failed} clip${failed===1?"":"s"}.`:"");
    return {added,failed};
  } finally {
    setBusy(false);
  }
};

function arrayBufferToBase64(buffer){
  const bytes=new Uint8Array(buffer);
  let binary="";
  const chunk=0x8000;
  for(let i=0;i<bytes.length;i+=chunk){
    binary+=String.fromCharCode(...bytes.subarray(i,Math.min(bytes.length,i+chunk)));
  }
  return btoa(binary);
}

window.skiTrainerPublishModel = async function(){
  if(!model || modelDatasetRevision!==datasetRevision) throw new Error("Train the current Supabase dataset before publishing the model.");

  let captured=null;
  await model.save(tf.io.withSaveHandler(async artifacts=>{
    captured=artifacts;
    return {
      modelArtifactsInfo:{
        dateSaved:new Date(),
        modelTopologyType:"JSON",
        modelTopologyBytes:JSON.stringify(artifacts.modelTopology||{}).length,
        weightSpecsBytes:JSON.stringify(artifacts.weightSpecs||[]).length,
        weightDataBytes:artifacts.weightData?.byteLength||0
      }
    };
  }));

  if(!captured?.weightData) throw new Error("Could not capture trained model weights.");

  const config={
    version:2,
    created:new Date().toISOString(),
    classNames:[...classNames],
    trickScores:{...trickScores},
    trainedSteps,
    trainedFeatureCount
  };

  const response=await fetch("/api/model",{
    method:"POST",
    headers:{"content-type":"application/json"},
    body:JSON.stringify({
      modelTopology:captured.modelTopology,
      weightSpecs:captured.weightSpecs,
      weightDataBase64:arrayBufferToBase64(captured.weightData),
      config
    })
  });

  const data=await response.json().catch(()=>({}));
  if(!response.ok) throw new Error(data.error||`Model publish failed (${response.status})`);
  return config;
};

window.skiTrainerExportModel = async function(){
  if(!model || modelDatasetRevision!==datasetRevision) throw new Error("Train the current dataset before exporting the model.");
  await model.save("downloads://ski-trick-model");
  const config={
    version:1,
    created:new Date().toISOString(),
    classNames,
    trickScores,
    trainedSteps,
    trainedFeatureCount
  };
  const blob=new Blob([JSON.stringify(config,null,2)],{type:"application/json"});
  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");
  a.href=url;
  a.download="ski-trick-model-config.json";
  a.click();
  setTimeout(()=>URL.revokeObjectURL(url),500);
  return config;
};
