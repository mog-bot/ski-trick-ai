  if(!payload?.modelTopology || !Array.isArray(payload.weightSpecs) || !payload.weightData){

  }

    weightData: payload.weightData
  }));

  if(model) model.dispose();

  modelDatasetRevision = datasetRevision;

function normalizeTrickName(label){
  return String(label || "")
    .trim()
    .toLowerCase()
    .replace(/[\s_-]+/g,"");
}

function scoreFor(label){
  const wanted = normalizeTrickName(label);
  for(const [name,points] of Object.entries(trickScores)){
    if(normalizeTrickName(name) === wanted){
      return Number(points) || 0;
    }
  }
  return 0;
}

function renderScores(){
  const holder = $("scoreList");
  if(!holder) return;

  for(const label of [...dataset.map(x=>x.label), ...classNames]){
    const exists = Object.keys(trickScores).some(
      k => normalizeTrickName(k) === normalizeTrickName(label)
    );
    if(!exists) trickScores[label] = 0;
  }

  const names = new Set([
    ...Object.keys(trickScores),
    ...dataset.map(x => x.label),
    ...classNames
  ]);

  const ordered = [...names].sort((a,b)=>a.localeCompare(b));

  holder.innerHTML = ordered.length
    ? ordered.map(label => `
      <div class="scoreRow">
        <div><strong>${esc(label)}</strong></div>
        <input
          type="number"
          min="0"
          step="1"
          value="${scoreFor(label)}"

function skiFromFoot(pose, side){
  const ankleIdx = side==="left" ? 27 : 28;
  const heelIdx = side==="left" ? 29 : 30;
  const toeIdx = side==="left" ? 31 : 32;
  const ankle = pose[ankleIdx], heel = pose[heelIdx], toe = pose[toeIdx];
  if(!ankle || !heel || !toe) return null;

  let dx = toe.x - heel.x, dy = toe.y - heel.y;
  const mag = Math.sqrt(dx*dx + dy*dy);
  if(mag < 1e-5) return null;
  dx /= mag; dy /= mag;

  const shoulderWidth = Math.max(
    Math.hypot(pose[11].x-pose[12].x, pose[11].y-pose[12].y),
    0.04
  );
  const halfLength = shoulderWidth * 1.8;
  const center = {
    x:(ankle.x + toe.x + heel.x)/3,
    y:(ankle.y + toe.y + heel.y)/3
  };

  return {
    center,
    tip:{x:center.x + dx*halfLength, y:center.y + dy*halfLength},
    tail:{x:center.x - dx*halfLength, y:center.y - dy*halfLength},
    angle:Math.atan2(dy,dx),
    visibility:Math.min(ankle.visibility ?? 0, heel.visibility ?? 0, toe.visibility ?? 0)
  };
}
